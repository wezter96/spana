<section class="spana-section" style="margin-top: 0">
  <div class="spana-section-header">
    <span class="spana-eyebrow">How it looks</span>
    <h2>A test is a typed TypeScript function</h2>
    <p>
      Tag elements with a <code>testID</code> once. Spana finds them on the web page, in the
      WebView of your iOS or Android app, and in your desktop app.
    </p>
  </div>

<div class="spana-proof">

```ts title="flows/login.flow.ts"
import { flow } from "spana-test";

export default flow("user can log in", async ({ app, expect }) => {
  await app.tap({ testID: "email-input" });
  await app.inputText("user@example.com");
  await app.tap({ testID: "login-button" });
  await expect({ testID: "home-screen" }).toBeVisible();
});

// spana test --platform web,android,ios
```

<section class="spana-section">
  <div class="spana-section-header">
    <span class="spana-eyebrow">Why spana</span>
    <h2>Less setup, faster runs, agent-ready</h2>
  </div>

  <div class="spana-benefits">
    <div class="spana-benefit">
      <h3>One suite, five targets</h3>
      <p>
        The same <code>.flow.ts</code> (or Gherkin <code>.feature</code>) file runs on Web, iOS,
        Android, Electron and Electrobun.
      </p>
    </div>
    <div class="spana-benefit">
      <h3>No Appium to install</h3>
      <p>
        Spana talks to the iOS and Android drivers directly. Use BrowserStack or Sauce Labs when you
        need real devices in the cloud.
      </p>
    </div>
    <div class="spana-benefit">
      <h3>Your agent can test for you</h3>
      <p>
        Run <code>spana mcp</code> and Claude Code, Codex or Cursor can see the screen, tap, check
        results, and save the steps as a real test file.
      </p>
    </div>
    <div class="spana-benefit">
      <h3>CI runs only what changed</h3>
      <p>
        <code>{"--changed-since origin/main"}</code> skips flows a pull request cannot affect. Parallel
        runs put the slowest flows first.
      </p>
    </div>
  </div>
</section>
